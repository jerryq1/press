### 后端 SSE 生成器与状态拉齐逻辑

#### [完整代码] [app/utils/sse_utils.py](file:///Users/jerry/Desktop/AI/rag_agent/app/utils/sse_utils.py) 中 `sse_generator` 实现部分：
```python
async def sse_generator(session_id: str, request: Request):
    """
    SSE 生成器，用于 FastAPI 的 StreamingResponse
    """
    print(f"[SSE] Generator started for session: {session_id}")
    stream_queue = get_sse_queue(session_id)
    if stream_queue is None:
        # 如果没有对应的队列，直接结束
        print(f"[SSE] Error: Queue not found for session {session_id}. Available sessions: {list(_session_stream.keys())}")
        return

    loop = asyncio.get_running_loop()
    try:
        # 发送连接建立信号
        print(f"[SSE] Sending ready signal for {session_id}")
        yield _sse_pack("ready", {})

        # 主动对齐当前任务的最新状态，防止连接建立延迟导致漏掉前期事件
        try:
            from app.utils.task_utils import (
                get_task_status,
                get_done_task_list,
                get_running_task_list,
                get_raw_done_task_list,
                get_raw_running_task_list
            )
            status = get_task_status(session_id)
            if status:
                current_progress = {
                    "status": status,
                    "done_list": get_done_task_list(session_id),
                    "running_list": get_running_task_list(session_id),
                    "raw_done_list": get_raw_done_task_list(session_id),
                    "raw_running_list": get_raw_running_task_list(session_id),
                }
                print(f"[SSE] Proactively pushing initial progress for {session_id}: {status}")
                yield _sse_pack("progress", current_progress)
        except Exception as ex:
            print(f"[SSE] Failed to send initial progress for {session_id}: {ex}")

        while True:
            # 若客户端断开，尽快退出
            if await request.is_disconnected():
                print(f"[SSE] Client disconnected: {session_id}")
                print("-----------------------断开连接--------------------")
                break

            try:
                # 使用 run_in_executor 避免阻塞 async 事件循环
                msg = await loop.run_in_executor(None, stream_queue.get, True, 1.0)
            except queue.Empty:
                # print(f"[SSE] Queue empty for {session_id}, waiting...")
                continue

            event = msg.get("event")
            data = msg.get("data")
            
            # print(f"[SSE] Yielding event {event} for {session_id}")

            # 特殊关闭事件
            if event == "__close__":
                print(f"[SSE] Closing signal received for {session_id}")
                break

            yield _sse_pack(event, data)
    except (asyncio.CancelledError, ConnectionResetError, BrokenPipeError):
        print(f"[SSE] Client disconnected (Cancelled/Reset/Pipe): {session_id}")
        # 生成器被取消/对端断开：静默退出
        return
    except Exception as e:
        print(f"[SSE] Exception in generator for {session_id}: {e}")
    finally:
        print(f"[SSE] Generator finished for {session_id}")
        # 清理资源
        remove_sse_queue(session_id)
```

### 后端 API 控制器与输出目录隔离

#### [完整代码] [app/api/file_import_service.py](file:///Users/jerry/Desktop/AI/rag_agent/app/api/file_import_service.py) 中核心 `/files` 与静态托管实现：
```python
@app.get("/files", summary="查询上传文件列表", description="获取doc目录下的所有上传文件，包括其处理状态与进度")
async def list_files():
    doc_dir = PROJECT_ROOT / "doc"
    if not doc_dir.exists():
        return {"code": 200, "files": []}
        
    file_list = []
    # 递归扫描doc下的所有文件
    for p in doc_dir.rglob("*"):
        if p.is_file() and not p.name.startswith("."):
            rel_path = p.relative_to(doc_dir)
            parts = rel_path.parts
            
            # 判断文件是在根目录，还是在以task_id命名的子目录下
            if len(parts) == 1:
                # 根目录文件：基于文件名生成稳定、唯一的task_id
                filename = parts[0]
                task_id = str(uuid.uuid5(uuid.NAMESPACE_DNS, filename))
            elif len(parts) == 2:
                # 子目录文件：子文件夹名即为task_id
                task_id = parts[0]
                filename = parts[1]
            else:
                # 更深层的嵌套
                task_id = parts[0]
                filename = parts[-1]
                
            local_dir = str(PROJECT_ROOT / "output" / task_id)
                
            status = get_task_status(task_id)
            # 如果内存里完全没有该任务的状态，则初始化为 pending，并把上传标记为完成
            if not status:
                status = "pending"
                update_task_status(task_id, "pending")
                add_done_task(task_id, "upload_file")
                
            file_list.append({
                "filename": filename,
                "task_id": task_id,
                "local_file_path": str(p.absolute()),
                "local_dir": local_dir,
                "status": status,
                "done_list": get_done_task_list(task_id),
                "running_list": get_running_task_list(task_id),
                "raw_done_list": get_raw_done_task_list(task_id),
                "raw_running_list": get_raw_running_task_list(task_id)
            })
            
    # 按文件名排序，方便前端稳定展示
    file_list.sort(key=lambda x: x["filename"])
    return {"code": 200, "files": file_list}
```

## 4. 复杂知识点 + 生活比喻

### 知识点 1：SSE (Server-Sent Events) 单向实时推流
*   **官方描述**：
    Server-Sent Events 是一种基于 HTTP 协议的服务器向浏览器单向发送数据的通信机制。客户端通过 HTML5 `EventSource` 发起请求，服务器保持长连接不断开，并使用 `text/event-stream` 媒体类型流式推送文本事件。
*   **生活比喻 (银行/餐厅场景)**：
    这就像去**美食广场点餐**。你付钱点完餐（POST 接口调用启动向量化）后，前台给你发了一个**号牌**（返回了 `task_id`），同时你只需要坐在位子上**看着上方的叫号显示屏幕**（EventSource 长连接）。当你的菜做好时，显示屏上会主动刷新亮起你的号（SSE 从后端实时推送进度包）。在这个过程中你不需要自己不断地跑到柜台去问菜好了没（不需要轮询，节省双方网络资源）。
*   **代码对应**：
    *   后端定义：[file_import_service.py](file:///Users/jerry/Desktop/AI/rag_agent/app/api/file_import_service.py#L70-L86) (接口路由与 StreamingResponse)
    *   推流生成器：[sse_utils.py](file:///Users/jerry/Desktop/AI/rag_agent/app/utils/sse_utils.py#L55-L106) (`sse_generator`)
    *   前端监听：[app.js](file:///Users/jerry/Desktop/AI/rag_agent/rag_agent_fronted/app.js#L306-L391) (`connectSSE` 函数)

### 知识点 2：FastAPI 的后台并发线程池 (BackgroundTasks)
*   **官方描述**：
    FastAPI 的 `BackgroundTasks` 可以在向客户端返回响应后在后台执行任务。如果任务函数是普通的 `def` 同步函数，FastAPI 会在独立的子线程（来自内部线程池）中异步执行它，不阻塞 asyncio 的主事件循环。
*   **生活比喻 (银行/餐厅场景)**：
    这就像你在银行柜台办理转账（发送 `/vectorize` 请求），柜员在系统里敲完单子，给你打印回执确认“已录入转账任务，请稍后查询”（立即响应 HTTP 200）。然后柜员把这个单子丢给后台的**清算部门**去慢慢处理（BackgroundTasks 异步启动工作流），自己立刻开始接待下一位排队的顾客。
*   **代码对应**：
    *   后台启动：[file_import_service.py](file:///Users/jerry/Desktop/AI/rag_agent/app/api/file_import_service.py#L162-L168) (`background_tasks.add_task(...)`)

### 知识点 3：多路由优先级匹配与单端口文件托管 (StaticFiles Mount)
*   **官方描述**：
    ASGI Web 框架中可以把其他的 ASGI 应用（如静态文件管理器 `StaticFiles`）挂载到特定的路径前缀下。FastAPI 按路由定义的顺序或优先级进行匹配，当特定路径都不匹配时，兜底传递给 mounted 路由。

## 5. 核心闭环总结

在这个系统中，**`task_id` (任务唯一 ID)** 是全局最核心的骨架字段，它将整条全生命周期通信链路串联为闭环：
1.  **物理存储层**：文件上传时生成 UUID 作为 `task_id`，文件保存在磁盘的 `doc/<task_id>/<filename>`。
2.  **状态映射层**：在内存任务字典 `_tasks_status`、`_tasks_done_list` 中，通过 `task_id` 唯一锁定了任务状态。
3.  **计算生成层**：向后台派发 `run_graph_task` 时传入 `task_id`。工作流流式运行时，每执行完一个节点，把 `task_id` 关联的节点名加入完成列表，实现进度追溯。
4.  **实时传输层**：前端建立 SSE 连接时，路径中携带 `/stream/{task_id}`。后端据此为该 `task_id` 建立消息队列；工作流中的更新触发 `push_to_session(task_id, ...)`，把消息定向路由分发至该长连接。
5.  **展示输出层**：前端收到事件，根据 `task_id` 匹配对应的卡片元素并渲染出正在运行或已完成的节点状态，最终关闭连接。

## 7. 异常情况处理说明

系统在各个交互节点上的异常防御和处理设计如下：

1.  **网络异常断开或客户端刷新**：
    *   **现象**：前端在任务中途刷新网页，或网络抖动导致 SSE 连接断开。
    *   **处理机制**：`sse_generator` 会在循环内调用 `request.is_disconnected()` 监听客户端心跳。一旦断开，触发 `finally` 块，自动执行 `remove_sse_queue(task_id)` 销毁队列，释放内存资源。由于任务执行在 `BackgroundTasks` 线程，后台的向量化流程仍然会继续运行直至完毕。
2.  **物理文件缺失**：
    *   **现象**：触发 `/vectorize` 时，磁盘上的 `local_file_path` 文件被意外删除。
    *   **处理机制**：接口入口处增加 `Path(local_file_path).exists()` 强校验。如不存在，直接抛出 `HTTPException(404)` 拦截并终止流程。
3.  **大模型/解析超时与崩溃**：
    *   **现象**：MinerU 轮询接口超时，或者 Milvus 导入时由于网络原因失败。
    *   **处理机制**：在 `run_graph_task` 内部通过 `try...except` 拦截异常，一旦捕获到任何错误，立即执行 `update_task_status(task_id, "failed")`，通过 SSE 队列向前端推送任务失败信号，并清理本会话长连接。

## 8. 常见问题 FAQ

#### Q1: 前端建立连接比后台任务启动慢，为什么页面不会漏掉“正在导入”的状态推送？
*   **解答**：因为我们实现了“**状态强行拉齐**”机制。在 SSE 生成器 `sse_generator` 初始化成功的一刹那，后端会先读取内存中存储的该 Task 最新状态并强制通过长连接往前端推送一条 `progress` 事件，然后再进入队列循环。这消除了前后端握手的时间差带来的竞态问题。

#### Q2: 如果我同时触发 10 个文件的向量化，后台会崩溃吗？
*   **解答**：不会崩溃。FastAPI 的 `BackgroundTasks` 在遇到同步 `def` 函数时，会将其放入一个预置大小的线程池（通常是核心数 * 5）中运行。超过池大小的任务会被排队等待，保证服务器不会因为并发量过高导致 CPU 或显存瞬间爆表。同时，由于各个任务的 `local_dir` 已经完成隔离（`output/task_id`），它们之间的文件读写互不干涉。

#### Q3: 为什么同一个端口（8080）下既能处理 REST API，又能提供网页静态服务？
*   **解答**：这得益于 FastAPI 对 ASGI 协议和 Starlette 的底层继承。当启动 Uvicorn 服务时，挂载在根路径下的 `StaticFiles` 应用作为子路由挂载器，会在其他具体的 API 路由（如 `/files`，`/upload`）不满足匹配条件时，将请求路由到 `StaticFiles` 处理，实现了多态单端口服务，极大地简化了本地环境下的运行配置。

## 9. 动手实验/验证方法

为了验证 SSE 单向推流是否能正确同步接收后端推送的状态快照，您可以按照以下步骤在本地运行一个简易的 Python 客户端脚本来进行独立测试。

### 实验：模拟前端 SSE 侦听客户端
1. 确保您的 FastAPI 服务已启动在 `127.0.0.1:8080`。
2. 在本地创建名为 `scratch/test_sse_client.py` 的脚本：
```python
import sys
import requests

def listen_to_task(task_id: str):
    url = f"http://127.0.0.1:8080/stream/{task_id}"
    print(f"开始订阅任务 [{task_id}] 的 SSE 事件流...")
    
    # 模拟 EventSource 发起请求，设置 stream=True 实现流式读取
    headers = {"Accept": "text/event-stream"}
    try:
        response = requests.get(url, headers=headers, stream=True, timeout=30)
        for line in response.iter_lines():
            if line:
                decoded_line = line.decode('utf-8')
                print(f"收到事件: {decoded_line}")
    except KeyboardInterrupt:
        print("\n手动终止监听。")
    except Exception as e:
        print(f"连接或传输异常: {e}")

if __name__ == "__main__":
    # 请传入您上传文件后在前端或接口中拿到的 task_id
    if len(sys.argv) < 2:
        print("使用说明: python test_sse_client.py <task_id>")
        sys.exit(1)
    listen_to_task(sys.argv[1])
```
3. 运行该脚本订阅一个已经开启向量化（或者处于 `processing` 状态）的 `task_id`，您将在终端直观地看到推送下来的 `event: ready`、`event: progress`（最新快照状态对齐数据）和后续每个节点完成时的通知报文。

## 10. 整体迁移指南

如果您想将这套**文件多格式上传 + SSE 状态主动同步监控**的架构迁移到您的其他全新项目中，可以遵循以下极简框架：

### 步骤 1：拷贝基础工具文件
将以下文件拷贝至新项目的通用模块中：
1. `sse_utils.py`：负责消息队列定义、打包、生成器路由流。
2. `task_utils.py`：定义内存任务数据字典，实现进度的 `add` 和 `status` 更新。

### 步骤 2：注册 Uvicorn/FastAPI 主程序路由
在新项目的入口文件中写入以下架构，即可实现单端口 API 与前端文件的双向托管：
```python
import uvicorn
from fastapi import FastAPI, BackgroundTasks
from fastapi.staticfiles import StaticFiles
from app.utils.sse_utils import create_sse_queue, sse_generator
from app.utils.task_utils import update_task_status, add_done_task

app = FastAPI()

# 1. 后台处理逻辑
def my_background_pipeline(task_id: str, file_path: str):
    update_task_status(task_id, "processing")
    # ... 节点执行逻辑
    add_done_task(task_id, "step_1")
    update_task_status(task_id, "completed")

# 2. 向量化启动路由
@app.post("/api/start/{task_id}")
async def start_task(task_id: str, file_path: str, bg: BackgroundTasks):
    bg.add_task(my_background_pipeline, task_id, file_path)
    return {"status": "triggered"}

# 3. SSE 长连接通道
@app.get("/api/stream/{task_id}")
async def stream_progress(task_id: str, request: Request):
    create_sse_queue(task_id)
    return StreamingResponse(sse_generator(task_id, request), media_type="text/event-stream")

# 4. 托管前端静态资源（必须放在接口路由定义的后方）
app.mount("/", StaticFiles(directory="frontend_dist", html=True), name="static")

if __name__ == "__main__":
    uvicorn.run(app, host="127.0.0.1", port=8080)
```
