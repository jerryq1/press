# LangChain函数封装以及流式调用

## 简单的流式调用

```python
from langchain_openai import ChatOpenAI

# 启用流式输出
llm = ChatOpenAI(
    model="gpt-4",
    streaming=True,  # 关键参数
    temperature=0.7
)

# 使用 stream 方法
for chunk in llm.stream("介绍一下LangChain"):
    print(chunk.content, end="", flush=True)
```

## 函数封装(示例代码)

```python
# 导入必要的库
from langchain_openai import ChatOpenAI
import os
from dotenv import load_dotenv
from langchain_core.exceptions import LangChainException

# 加载.env文件中的环境变量（指定编码，避免中文乱码）
load_dotenv(encoding='utf-8')

# 配置日志（可选，便于调试）
import logging

logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')
logger = logging.getLogger(__name__)


def init_llm_client() -> ChatOpenAI:
    """
    初始化LLM客户端（封装成函数，提高复用性）

    Returns:
        ChatOpenAI: 初始化后的LLM客户端实例
    """
    # 1. 读取环境变量并做非空校验
    api_key = os.getenv("huoshan_api_key")
    print(f"os.getenv {api_key}")

    if not api_key:
        raise ValueError("环境变量 huoshan_api_key 未配置，请检查.env文件")

    # 2. 初始化LLM客户端（参数命名规范，添加注释）
    llm = ChatOpenAI(
        model="deepseek-v3-2-251201",  # 模型名称
        api_key=api_key,  # API密钥
        base_url="https://ark.cn-beijing.volces.com/api/v3",  # 阿里云兼容接口地址
        temperature=0.7,  # 可选：添加温度参数，控制输出随机性
        max_tokens=2048  # 可选：限制输出长度，避免超限
    )
    return llm


def main():
    """主函数：封装核心逻辑，符合Python工程化规范"""
    try:
        # 初始化客户端
        llm = init_llm_client()
        logger.info("LLM客户端初始化成功")

        # 调用模型（问题用变量存储，提高可读性）
        question = "你是谁"
        response = llm.invoke(question)

        # 格式化输出结果（而非直接打印原始对象）
        logger.info(f"问题：{question}")
        logger.info(f"回答：{response.content}")

        print("====================以下是流式输出,另一种调用方式")
        print("*" * 50)
        responseStream = llm.stream("介绍下langchain，300字以内")
        for chunk in responseStream:
            print(chunk.content,end="")
    # 捕获具体异常（而非宽泛的Exception）
    except ValueError as e:
        logger.error(f"配置错误：{str(e)}")
    except LangChainException as e:
        logger.error(f"模型调用失败：{str(e)}")
    except Exception as e:
        logger.error(f"未知错误：{str(e)}")


# 脚本入口（符合Python规范，避免导入时执行代码）
if __name__ == "__main__":
    main()
```

## 函数封装(TS版本参考)

```typescript
// 导入必要的库
import { ChatOpenAI } from "@langchain/openai";
import dotenv from "dotenv";
import { LangChainException } from "@langchain/core/exceptions";
import { BaseCallbackHandler } from "@langchain/core/callbacks/base";
import * as readline from "readline";

// 加载.env文件中的环境变量
dotenv.config({ encoding: "utf-8" });

// 配置日志
const logger = {
  info: (message: string) => console.log(`[${new Date().toISOString()}] INFO - ${message}`),
  error: (message: string) => console.error(`[${new Date().toISOString()}] ERROR - ${message}`),
};

/**
 * 初始化LLM客户端（封装成函数，提高复用性）
 * @returns {ChatOpenAI} 初始化后的LLM客户端实例
 */
function initLLMClient(): ChatOpenAI {
  // 1. 读取环境变量并做非空校验
  const apiKey = process.env.HUOSHAN_API_KEY;
  console.log(`process.env.HUOSHAN_API_KEY: ${apiKey}`);

  if (!apiKey) {
    throw new Error("环境变量 HUOSHAN_API_KEY 未配置，请检查.env文件");
  }

  // 2. 初始化LLM客户端（参数命名规范，添加注释）
  const llm = new ChatOpenAI({
    model: "deepseek-v3-2-251201", // 模型名称
    apiKey: apiKey, // API密钥
    configuration: {
      baseURL: "https://ark.cn-beijing.volces.com/api/v3", // 火山引擎兼容接口地址
    },
    temperature: 0.7, // 控制输出随机性
    maxTokens: 2048, // 限制输出长度，避免超限
  });

  return llm;
}

/**
 * 主函数：封装核心逻辑，符合TypeScript工程化规范
 */
async function main(): Promise<void> {
  try {
    // 初始化客户端
    const llm = initLLMClient();
    logger.info("LLM客户端初始化成功");

    // 调用模型（问题用变量存储，提高可读性）
    const question = "你是谁";
    const response = await llm.invoke(question);

    // 格式化输出结果
    logger.info(`问题：${question}`);
    logger.info(`回答：${response.content as string}`);

    console.log("====================以下是流式输出,另一种调用方式");
    console.log("*".repeat(50));

    // 流式输出
    const stream = await llm.stream("介绍下langchain，300字以内");
    for await (const chunk of stream) {
      process.stdout.write(chunk.content as string);
    }
    console.log(); // 输出完成后换行

  } catch (error) {
    // 捕获具体异常
    if (error instanceof Error) {
      if (error.message.includes("未配置")) {
        logger.error(`配置错误：${error.message}`);
      } else if (error instanceof LangChainException) {
        logger.error(`模型调用失败：${error.message}`);
      } else {
        logger.error(`未知错误：${error.message}`);
      }
    } else {
      logger.error(`未知错误：${String(error)}`);
    }
  }
}

// 脚本入口（符合TypeScript规范，避免导入时执行代码）
if (require.main === module) {
  main();
}

export { initLLMClient, main };
```
