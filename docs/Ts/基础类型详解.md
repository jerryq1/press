## 1. 基本类型：number、string、boolean

### number - 数字类型

数字类型包括整数和浮点数，同时也支持二进制、八进制、十六进制表示。

```typescript
// 基础数字类型
let age: number = 25;
let price: number = 99.99;
let temperature: number = -5.5;

// 进制表示
let decimal: number = 42;           // 十进制
let hex: number = 0xff;             // 十六进制，255
let binary: number = 0b1010;        // 二进制，10
let octal: number = 0o744;          // 八进制，484

// 特殊数值
let infinity: number = Infinity;
let negativeInfinity: number = -Infinity;
let notANumber: number = NaN;
```

### string - 字符串类型

字符串类型可以使用单引号、双引号或模板字符串（反引号）定义。

```typescript
// 字符串定义方式
let singleQuote: string = '使用单引号';
let doubleQuote: string = "使用双引号";
let template: string = `使用模板字符串`;

// 模板字符串特性 - 支持表达式插值和多行文本
let firstName: string = "张";
let lastName: string = "三";
let fullName: string = `${firstName}${lastName}`;  // "张三"

let multiLine: string = `
  第一行
  第二行
  第三行
`;

// 字符串拼接
let greeting: string = "你好，" + fullName + "！";
let templateGreeting: string = `你好，${fullName}！`;
```

### boolean - 布尔类型

布尔类型只有两个值：`true` 和 `false`。

```typescript
// 基础布尔值
let isActive: boolean = true;
let isCompleted: boolean = false;
let isEnabled: boolean = true;

// 布尔表达式
let isGreater: boolean = 10 > 5;     // true
let isEqual: boolean = 10 === 5;     // false
let isNotEmpty: boolean = "hello".length > 0;  // true

// 条件判断中的布尔值
function checkStatus(status: boolean): string {
    return status ? "激活" : "未激活";
}
```

## 2. 数组和元组：Array、Tuple

### Array - 数组类型

TypeScript 中的数组有两种定义方式，可以存储相同类型的元素。

```typescript
// 方式一：类型 + 方括号
let numbers: number[] = [1, 2, 3, 4, 5];
let strings: string[] = ["a", "b", "c", "d"];
let booleans: boolean[] = [true, false, true];

// 方式二：泛型 Array<类型>
let numbersGeneric: Array<number> = [1, 2, 3, 4, 5];
let stringsGeneric: Array<string> = ["a", "b", "c", "d"];

// 只读数组
let readonlyNumbers: readonly number[] = [1, 2, 3];
// readonlyNumbers[0] = 10;  // 错误：只读数组不能修改

// 混合类型数组（使用联合类型）
let mixed: (number | string)[] = [1, "two", 3, "four", 5];

// 数组常用操作
let fruits: string[] = ["苹果", "香蕉", "橙子"];
fruits.push("葡萄");           // 添加元素
let firstFruit: string = fruits[0];  // 访问元素
let fruitCount: number = fruits.length;  // 数组长度
```

### Tuple - 元组类型

元组允许表示一个已知元素数量和类型的数组，各元素的类型不必相同。

```typescript
// 基础元组定义
let person: [string, number] = ["李四", 30];
let rgb: [number, number, number] = [255, 128, 0];
let httpResponse: [number, string] = [200, "成功"];

// 访问元组元素
console.log(person[0]);  // "李四"
console.log(person[1]);  // 30
console.log(rgb[0]);     // 255

// 元组可选元素
let optionalTuple: [string, number?] = ["可选"];
let tupleWithOptional: [string, number?, boolean?] = ["hello", 42];

// 元组剩余元素
let restTuple: [string, ...number[]] = ["开始", 1, 2, 3, 4, 5];

// 元组解构
let [name, age] = person;
console.log(`${name} 今年 ${age} 岁`);

// 元组实际应用场景
function getPerson(): [string, number] {
    return ["王五", 25];
}

let [personName, personAge] = getPerson();

// 带标签的元组（TypeScript 4.0+）
let labeledTuple: [name: string, age: number, email?: string] = ["张三", 30];
```

## 3. any 和 unknown：任意类型

### any - 任意类型

`any` 类型会关闭 TypeScript 的类型检查，让变量可以接受任何类型的值。**应尽量避免使用**。

```typescript
// any 类型的使用
let flexible: any = "可以是任何类型";
console.log(flexible.toUpperCase());  // 可以调用字符串方法

flexible = 42;
console.log(flexible.toFixed(2));     // 可以调用数字方法

flexible = true;
console.log(flexible.valueOf());      // 可以调用布尔方法

flexible = { key: "value" };
console.log(flexible.key);            // 可以访问对象属性

// any 在数组中的应用
let anyArray: any[] = [1, "string", true, { id: 1 }, [1, 2, 3]];

// 使用 any 的风险 - 不会报错，但可能在运行时出错
let risky: any = "hello";
risky.someMethod();  // 编译通过，但运行时会报错
```

### unknown - 未知类型

`unknown` 是类型安全的 `any`，使用前必须进行类型检查。

```typescript
// unknown 类型定义
let uncertain: unknown = "未知类型";
uncertain = 100;
uncertain = true;
uncertain = { data: "some data" };

// unknown 不能直接使用
// console.log(uncertain.toUpperCase());  // 错误：类型检查阻止

// 需要进行类型守卫
if (typeof uncertain === "string") {
    console.log(uncertain.toUpperCase());  // 安全使用
}

if (typeof uncertain === "number") {
    console.log(uncertain.toFixed(2));     // 安全使用
}

// 使用 instanceof 检查
class MyClass {
    method() {
        console.log("方法调用");
    }
}

let obj: unknown = new MyClass();
if (obj instanceof MyClass) {
    obj.method();  // 安全调用
}

// 使用类型断言（谨慎使用）
if ((uncertain as { data: string }).data) {
    console.log((uncertain as { data: string }).data);
}

// 自定义类型守卫
function isString(value: unknown): value is string {
    return typeof value === "string";
}

if (isString(uncertain)) {
    console.log(uncertain.toUpperCase());  // 类型守卫确保安全
}
```

### any vs unknown 对比

| 特性 | any | unknown |
|------|-----|---------|
| 类型安全 | ❌ 不安全 | ✅ 安全 |
| 访问属性 | ✅ 可以 | ❌ 需要检查 |
| 调用方法 | ✅ 可以 | ❌ 需要检查 |
| 赋值给其他类型 | ✅ 可以 | ❌ 需要断言 |
| 推荐使用 | ❌ 不推荐 | ✅ 推荐 |

```typescript
// any vs unknown 示例
let anyVar: any = "hello";
let unknownVar: unknown = "world";

// any 可以直接赋值给其他类型
let str1: string = anyVar;  // ✅ 允许
let num1: number = anyVar;  // ✅ 允许（危险）

// unknown 不能直接赋值给其他类型
let str2: string = unknownVar;  // ❌ 错误
let str3: string = unknownVar as string;  // ✅ 需要类型断言
```

## 4. void、null、undefined

### void - 无返回值

`void` 通常用于表示函数没有返回值。

```typescript
// 函数没有返回值
function logMessage(message: string): void {
    console.log(message);
    // 没有 return 语句
}

function doNothing(): void {
    return;  // 可以返回 undefined
}

// 箭头函数中的 void
const logError = (error: string): void => {
    console.error(error);
};

// void 类型的变量（不常用）
let unusable: void = undefined;
// unusable = null;  // 如果 strictNullChecks 为 false 可以赋值 null

// 回调函数中使用 void
function forEach<T>(items: T[], callback: (item: T) => void): void {
    for (const item of items) {
        callback(item);
    }
}
```

### null - 空值

`null` 表示一个空值或不存在的值。

```typescript
// null 类型
let empty: null = null;

// 实际应用场景
let userInput: string | null = null;  // 可能为 null 的变量

function findUser(id: number): { name: string } | null {
    if (id > 0) {
        return { name: "张三" };
    }
    return null;  // 未找到用户
}

// 处理可能为 null 的值
let user = findUser(0);
if (user !== null) {
    console.log(user.name);
}

// 使用可选链操作符（?.）处理 null
let userName = user?.name;  // 如果 user 为 null，返回 undefined
```

### undefined - 未定义

`undefined` 表示变量未被赋值。

```typescript
// undefined 类型
let notDefined: undefined = undefined;

// 未初始化的变量默认为 undefined
let uninitialized: string;
console.log(uninitialized);  // undefined

// 函数可能返回 undefined
function getUserName(): string | undefined {
    // 可能返回 undefined
    return undefined;
}

// 可选参数默认为 undefined
function greet(name?: string): string {
    if (name === undefined) {
        return "你好，访客";
    }
    return `你好，${name}`;
}

// 对象属性可能为 undefined
interface Config {
    url: string;
    timeout?: number;  // 可选属性，可能为 undefined
}

let config: Config = { url: "https://api.example.com" };
console.log(config.timeout);  // undefined
```

### null 和 undefined 的区别与处理

```typescript
// null 和 undefined 的区别
let a: null = null;
let b: undefined = undefined;

// 严格相等比较
console.log(null === undefined);  // false
console.log(null == undefined);   // true

// 类型守卫
function processValue(value: string | null | undefined): string {
    if (value === null) {
        return "值为 null";
    }
    if (value === undefined) {
        return "值为 undefined";
    }
    return `值为: ${value}`;
}

// 空值合并操作符（??）
let input: string | null | undefined = null;
let result = input ?? "默认值";  // 只在 null 或 undefined 时使用默认值
console.log(result);  // "默认值"
```

## 5. never：永不存在的值的类型

`never` 类型表示永远不会发生的类型，用于函数永远不会返回或总是抛出异常的情况。

### 基础用法

```typescript
// 抛出异常的函数 - 永远不会正常返回
function throwError(message: string): never {
    throw new Error(message);
}

// 无限循环的函数 - 永远不会结束
function infiniteLoop(): never {
    while (true) {
        // 永远不会结束
    }
}

// 程序退出函数
function exitProcess(code: number): never {
    process.exit(code);
}
```

### 类型守卫中的应用

```typescript
// never 在 exhaustive 检查中的应用
type Shape = "circle" | "square" | "triangle";

function getArea(shape: Shape, size: number): number {
    switch (shape) {
        case "circle":
            return Math.PI * size * size;
        case "square":
            return size * size;
        case "triangle":
            return (Math.sqrt(3) / 4) * size * size;
        default:
            // 这里 shape 的类型是 never，因为所有情况都已覆盖
            const exhaustiveCheck: never = shape;
            return exhaustiveCheck;
    }
}
```

### never 在类型系统中的作用

```typescript
// never 是其他类型的子类型
type T1 = never extends string ? true : false;  // true
type T2 = never extends number ? true : false;  // true

// 联合类型中的 never 会被忽略
type T3 = string | never;  // string
type T4 = number | never;  // number

// never 的实际应用 - 确保类型检查的完整性
interface User {
    type: "admin" | "user" | "guest";
    name: string;
}

function processUser(user: User): string {
    switch (user.type) {
        case "admin":
            return `管理员 ${user.name}`;
        case "user":
            return `普通用户 ${user.name}`;
        case "guest":
            return `访客 ${user.name}`;
        default:
            // 如果将来添加了新的用户类型，这里会报错
            const _exhaustiveCheck: never = user.type;
            return _exhaustiveCheck;
    }
}
```

### never 与 void 的区别

| 特性 | void | never |
|------|------|-------|
| 返回值 | 可以返回 undefined | 永远不返回值 |
| 函数行为 | 正常返回 | 总是抛出异常或无限循环 |
| 使用场景 | 函数没有返回值 | 函数永远不会返回 |

```typescript
// void vs never 对比
function voidFunction(): void {
    return;  // 可以正常返回
}

function neverFunction(): never {
    throw new Error("永远不返回");
    // 这里的代码永远不会执行
}
```

## 6. 枚举（enum）

枚举用于定义一组命名常量，TypeScript 支持数字枚举和字符串枚举。

### 数字枚举

```typescript
// 基础数字枚举 - 默认从 0 开始
enum Direction {
    Up,      // 0
    Down,    // 1
    Left,    // 2
    Right    // 3
}

console.log(Direction.Up);    // 0
console.log(Direction[0]);    // "Up" - 反向映射

// 自定义起始值
enum Status {
    Pending = 1,
    Approved = 2,
    Rejected = 3
}

// 混合使用
enum Mixed {
    A = 1,
    B,      // 2
    C = 5,
    D,      // 6
    E = 10
}

console.log(Mixed.A);  // 1
console.log(Mixed.B);  // 2
console.log(Mixed.C);  // 5
console.log(Mixed.D);  // 6
```

### 字符串枚举

```typescript
// 字符串枚举 - 每个成员都必须用字符串字面量初始化
enum Color {
    Red = "红色",
    Green = "绿色",
    Blue = "蓝色"
}

// 字符串枚举没有反向映射
console.log(Color.Red);     // "红色"
console.log(Color["红色"]);  // undefined - 没有反向映射

// 混合字符串和数字枚举（不推荐）
enum MixedEnum {
    A = "string",
    B = 100,
    C = "another"
}
```

### const 枚举

```typescript
// const 枚举 - 编译后会被移除，直接内联
const enum LogLevel {
    Info = "INFO",
    Warning = "WARNING",
    Error = "ERROR"
}

// 编译后直接变成字符串常量
let level = LogLevel.Info;  // 编译后: let level = "INFO";
```

### 枚举的实际应用

```typescript
// 1. 状态管理
enum OrderStatus {
    Pending = "待处理",
    Processing = "处理中",
    Shipped = "已发货",
    Delivered = "已送达",
    Cancelled = "已取消"
}

interface Order {
    id: number;
    status: OrderStatus;
    total: number;
}

function getOrderStatusMessage(status: OrderStatus): string {
    switch (status) {
        case OrderStatus.Pending:
            return "订单已创建，等待处理";
        case OrderStatus.Processing:
            return "订单正在处理中";
        case OrderStatus.Shipped:
            return "订单已发货";
        case OrderStatus.Delivered:
            return "订单已送达";
        case OrderStatus.Cancelled:
            return "订单已取消";
        default:
            return "未知状态";
    }
}

// 2. API 响应状态
enum HttpStatus {
    OK = 200,
    Created = 201,
    BadRequest = 400,
    Unauthorized = 401,
    NotFound = 404,
    InternalServerError = 500
}

// 3. 权限管理
enum Permission {
    Read = 1 << 0,   // 1
    Write = 1 << 1,  // 2
    Execute = 1 << 2, // 4
    Delete = 1 << 3   // 8
}

// 组合权限
let userPermissions: Permission = Permission.Read | Permission.Write;
console.log(userPermissions);  // 3

// 检查权限
function hasPermission(permissions: Permission, check: Permission): boolean {
    return (permissions & check) === check;
}

console.log(hasPermission(userPermissions, Permission.Read));    // true
console.log(hasPermission(userPermissions, Permission.Execute)); // false
```

## 7. 综合示例

### 用户管理系统示例

```typescript
// 枚举定义
enum Role {
    Admin = "管理员",
    User = "普通用户",
    Guest = "访客"
}

enum UserStatus {
    Active = "active",
    Inactive = "inactive",
    Banned = "banned"
}

// 类型别名
type UserId = number;
type Email = string | null;

// 接口定义
interface User {
    id: UserId;
    name: string;
    email: Email;
    role: Role;
    status: UserStatus;
    tags: string[];
    metadata: unknown;
    createdAt: Date;
}

// 元组类型
type UserResponse = [User | null, string | null];  // [用户数据, 错误信息]

// 用户服务类
class UserService {
    private users: Map<UserId, User> = new Map();
    
    // 创建用户
    createUser(name: string, email: Email, role: Role = Role.User): UserResponse {
        try {
            const id = Date.now();
            const newUser: User = {
                id,
                name,
                email,
                role,
                status: UserStatus.Active,
                tags: [],
                metadata: {},
                createdAt: new Date()
            };
            
            this.users.set(id, newUser);
            return [newUser, null];
        } catch (error) {
            return [null, error instanceof Error ? error.message : "创建用户失败"];
        }
    }
    
    // 获取用户
    getUser(id: UserId): User | null {
        return this.users.get(id) ?? null;
    }
    
    // 更新用户状态
    updateUserStatus(id: UserId, status: UserStatus): void {
        const user = this.users.get(id);
        if (user) {
            user.status = status;
        }
    }
    
    // 添加用户标签
    addUserTag(id: UserId, tag: string): void {
        const user = this.users.get(id);
        if (user) {
            user.tags.push(tag);
        }
    }
    
    // 更新用户元数据
    updateUserMetadata(id: UserId, metadata: unknown): void {
        const user = this.users.get(id);
        if (user) {
            user.metadata = metadata;
        }
    }
    
    // 获取所有用户
    getAllUsers(): User[] {
        return Array.from(this.users.values());
    }
    
    // 获取管理员列表
    getAdminUsers(): User[] {
        return this.getAllUsers().filter(user => user.role === Role.Admin);
    }
    
    // 类型守卫：检查是否为活跃用户
    isActiveUser(user: User | null): user is User {
        return user !== null && user.status === UserStatus.Active;
    }
    
    // 处理用户信息（使用 never 进行完整性检查）
    processUserInfo(user: User): string {
        const statusMap: Record<UserStatus, string> = {
            [UserStatus.Active]: "活跃用户",
            [UserStatus.Inactive]: "非活跃用户",
            [UserStatus.Banned]: "已被封禁"
        };
        
        const roleMap: Record<Role, string> = {
            [Role.Admin]: "管理员权限",
            [Role.User]: "普通用户权限",
            [Role.Guest]: "访客权限"
        };
        
        // 处理 metadata（unknown 类型）
        let metadataInfo = "";
        if (user.metadata && typeof user.metadata === "object") {
            metadataInfo = `，元数据: ${JSON.stringify(user.metadata)}`;
        }
        
        return `${user.name} (${user.email ?? "无邮箱"}) - ${statusMap[user.status]} - ${roleMap[user.role]}${metadataInfo}`;
    }
}

// 使用示例
const userService = new UserService();

// 创建用户
const [user1, error1] = userService.createUser("张三", "zhangsan@example.com", Role.Admin);
const [user2, error2] = userService.createUser("李四", null, Role.User);
const [user3, error3] = userService.createUser("王五", "wangwu@example.com", Role.Guest);

if (user1) {
    userService.addUserTag(user1.id, "VIP");
    userService.updateUserMetadata(user1.id, { loginCount: 10, lastLogin: "2024-01-15" });
}

if (user2) {
    userService.updateUserStatus(user2.id, UserStatus.Inactive);
}

// 获取用户信息
const foundUser = userService.getUser(user1?.id ?? 0);
if (userService.isActiveUser(foundUser)) {
    console.log("活跃用户:", foundUser.name);
}

// 处理用户信息
userService.getAllUsers().forEach(user => {
    console.log(userService.processUserInfo(user));
});

// 获取管理员列表
const admins = userService.getAdminUsers();
console.log(`管理员数量: ${admins.length}`);
```

### 泛型与类型守卫示例

```typescript
// 泛型工具函数
function isNotNull<T>(value: T | null): value is T {
    return value !== null;
}

function isNotUndefined<T>(value: T | undefined): value is T {
    return value !== undefined;
}

function isDefined<T>(value: T | null | undefined): value is T {
    return value !== null && value !== undefined;
}

// 安全访问对象属性
function safeAccess<T extends object, K extends keyof T>(
    obj: T | null | undefined,
    key: K
): T[K] | undefined {
    return obj?.[key];
}

// 类型断言函数
function assertNever(value: never): never {
    throw new Error(`Unexpected value: ${value}`);
}

// 使用示例
const user = userService.getUser(user1?.id ?? 0);
if (isDefined(user)) {
    console.log(user.name);
}

const userName = safeAccess(user, "name");
console.log(userName);  // "张三" 或 undefined
```

| 类型 | 说明 | 示例代码 | 使用场景 |
|------|------|----------|----------|
| `number` | 数字类型，支持整数和浮点数 | `let age: number = 25` | 年龄、价格、ID、计数器 |
| `string` | 字符串类型 | `let name: string = "张三"` | 用户名、描述、文本数据 |
| `boolean` | 布尔类型 | `let isActive: boolean = true` | 状态开关、条件判断 |
| `Array<T>` | 数组类型 | `let list: number[] = [1, 2, 3]` | 列表数据、集合 |
| `Tuple` | 元组类型 | `let x: [string, number] = ["hi", 10]` | 固定结构的数据，如坐标、键值对 |
| `any` | 任意类型（不安全） | `let data: any = "anything"` | 动态内容、第三方库（不推荐） |
| `unknown` | 未知类型（安全） | `let data: unknown = "safe"` | 动态内容、API 响应（推荐） |
| `void` | 无返回值 | `function log(): void {}` | 不返回值的函数 |
| `null` | 空值 | `let empty: null = null` | 明确表示空值 |
| `undefined` | 未定义 | `let undef: undefined = undefined` | 未初始化的变量 |
| `never` | 永不返回 | `function error(): never { throw ... }` | 异常处理、完整性检查 |
| `enum` | 枚举类型 | `enum Color { Red, Green }` | 一组相关的常量 |

## 9. 最佳实践建议

### 1. 类型选择原则

```typescript
// ✅ 推荐：使用 unknown 而不是 any
function processApiResponse(response: unknown) {
    if (typeof response === 'object' && response !== null) {
        // 安全处理
    }
}

// ❌ 避免：使用 any
function processApiResponseBad(response: any) {
    // 失去了类型安全
}
```

### 2. 使用联合类型处理多种可能

```typescript
// ✅ 推荐：明确可能的类型
type InputValue = string | number | null | undefined;

function processInput(value: InputValue): string {
    if (value === null || value === undefined) {
        return "空值";
    }
    return String(value);
}
```

### 3. 利用类型守卫

```typescript
// ✅ 推荐：创建可复用的类型守卫
function isString(value: unknown): value is string {
    return typeof value === "string";
}

function isNumber(value: unknown): value is number {
    return typeof value === "number";
}

function processValue(value: unknown): string {
    if (isString(value)) {
        return value.toUpperCase();
    }
    if (isNumber(value)) {
        return value.toFixed(2);
    }
    return "未知类型";
}
```

### 4. 枚举使用建议

```typescript
// ✅ 推荐：字符串枚举更易调试
enum Status {
    Active = "ACTIVE",
    Inactive = "INACTIVE"
}

// ✅ 推荐：const 枚举减少编译后代码
const enum Direction {
    Up = "UP",
    Down = "DOWN"
}

// ⚠️ 注意：数字枚举的反向映射可能导致意外行为
enum Role {
    Admin = 1,
    User = 2
}
console.log(Role[1]);  // "Admin" - 反向映射存在
```

### 5. 严格空值检查

```typescript
// 在 tsconfig.json 中启用 strictNullChecks
{
    "compilerOptions": {
        "strictNullChecks": true
    }
}

// 明确处理可能为 null/undefined 的值
function getUserName(user: { name?: string | null }): string {
    // 使用可选链和空值合并
    return user.name?.toUpperCase() ?? "匿名用户";
}
```

### 6. never 类型的实际应用

```typescript
// 用于确保 switch 语句的完整性
type Action = "create" | "read" | "update" | "delete";

function handleAction(action: Action): string {
    switch (action) {
        case "create":
            return "创建";
        case "read":
            return "读取";
        case "update":
            return "更新";
        case "delete":
            return "删除";
        default:
            // 如果未来添加新的 action 类型，这里会报错
            const _exhaustiveCheck: never = action;
            return _exhaustiveCheck;
    }
}
```

### 7. 类型声明建议

```typescript
// ✅ 推荐：使用 type 或 interface 定义复杂类型
interface User {
    id: number;
    name: string;
    email?: string;  // 可选属性
}

// ✅ 推荐：使用字面量类型限制可能的值
type HttpMethod = "GET" | "POST" | "PUT" | "DELETE";

// ✅ 推荐：使用 Record 工具类型
type UserMap = Record<number, User>;
```
