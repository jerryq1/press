## 1. 函数类型注解

### 1.1 函数声明类型注解

为函数参数和返回值添加类型注解。

```typescript
// 基础函数类型注解
function greet(name: string): string {
    return `Hello, ${name}!`;
}

// 多个参数
function add(a: number, b: number): number {
    return a + b;
}

// 无返回值
function logMessage(message: string): void {
    console.log(message);
}
```

### 1.2 函数表达式类型注解

为函数表达式添加类型注解。

```typescript
// 方式一：变量类型注解
const greet: (name: string) => string = function(name) {
    return `Hello, ${name}!`;
};

// 方式二：箭头函数
const add: (a: number, b: number) => number = (a, b) => a + b;

// 方式三：使用 type 定义函数类型
type GreetFunction = (name: string) => string;
const greet2: GreetFunction = (name) => `Hello, ${name}!`;
```

### 1.3 对象方法类型注解

为对象中的方法添加类型注解。

```typescript
// 对象方法注解
const user = {
    name: "鱼皮",
    greet(name: string): string {
        return `Hello, ${name}!`;
    },
    add: (a: number, b: number): number => a + b
};

// 接口中的方法
interface Person {
    name: string;
    greet(name: string): string;
    add(a: number, b: number): number;
}
```

### 1.4 函数类型作为参数和返回值

```typescript
// 函数作为参数
function execute(callback: (result: string) => void): void {
    callback("执行完成");
}

// 函数作为返回值
function createAdder(x: number): (y: number) => number {
    return function(y: number): number {
        return x + y;
    };
}

const add5 = createAdder(5);
console.log(add5(3)); // 8
```

## 2. 可选参数和默认参数

### 2.1 可选参数

使用 `?` 标记参数为可选，可选参数必须放在必选参数之后。

```typescript
// 可选参数
function greet(name: string, greeting?: string): string {
    if (greeting) {
        return `${greeting}, ${name}!`;
    }
    return `Hello, ${name}!`;
}

console.log(greet("鱼皮"));        // "Hello, 鱼皮!"
console.log(greet("鱼皮", "Hi"));   // "Hi, 鱼皮!"

// 多个可选参数
function createUser(name: string, age?: number, email?: string): object {
    return {
        name,
        age,
        email
    };
}

// 可选参数类型可以为 undefined
function process(value?: string): void {
    // value 的类型是 string | undefined
    console.log(value?.toUpperCase());
}
```

### 2.2 默认参数

为参数提供默认值，当参数未传递或为 undefined 时使用默认值。

```typescript
// 基础默认参数
function createUser(name: string, age: number = 18): object {
    return { name, age };
}

console.log(createUser("鱼皮"));      // { name: "鱼皮", age: 18 }
console.log(createUser("鱼皮", 20));   // { name: "鱼皮", age: 20 }

// 默认参数可以放在可选参数之后
function request(url: string, method: string = "GET", timeout?: number): void {
    console.log(`${method} ${url}, timeout: ${timeout ?? 3000}`);
}

// 使用表达式作为默认值
function getDefaultAge(): number {
    return 18;
}

function createUserWithDefault(name: string, age: number = getDefaultAge()): object {
    return { name, age };
}
```

### 2.3 可选参数和默认参数的结合

```typescript
// 可选参数和默认参数混合使用
function greetUser(
    name: string,
    greeting?: string,
    punctuation: string = "!"
): string {
    const greet = greeting || "Hello";
    return `${greet}, ${name}${punctuation}`;
}

console.log(greetUser("鱼皮"));                    // "Hello, 鱼皮!"
console.log(greetUser("鱼皮", "Hi"));              // "Hi, 鱼皮!"
console.log(greetUser("鱼皮", "Hi", "~"));         // "Hi, 鱼皮~"
```

## 3. 剩余参数

使用 `...` 语法接收任意数量的参数，剩余参数必须是数组类型，且必须放在参数列表的最后。

### 3.1 基础剩余参数

```typescript
// 接收任意数量的数字参数
function sumAll(...numbers: number[]): number {
    return numbers.reduce((total, num) => total + num, 0);
}

console.log(sumAll(1, 2, 3));        // 6
console.log(sumAll(10, 20, 30, 40)); // 100

// 接收任意数量的字符串参数
function joinStrings(separator: string, ...strings: string[]): string {
    return strings.join(separator);
}

console.log(joinStrings("-", "a", "b", "c")); // "a-b-c"
```

### 3.2 剩余参数与普通参数结合

```typescript
// 普通参数在前，剩余参数在后
function logWithPrefix(prefix: string, ...messages: string[]): void {
    messages.forEach(msg => {
        console.log(`[${prefix}] ${msg}`);
    });
}

logWithPrefix("INFO", "开始执行", "处理数据", "完成执行");

// 多个普通参数
function createUserList(first: string, second: string, ...others: string[]): string[] {
    return [first, second, ...others];
}
```

### 3.3 剩余参数的类型约束

```typescript
// 剩余参数可以是联合类型
function processItems(...items: (string | number)[]): void {
    items.forEach(item => {
        if (typeof item === "string") {
            console.log(`字符串: ${item.toUpperCase()}`);
        } else {
            console.log(`数字: ${item.toFixed(2)}`);
        }
    });
}

processItems("hello", 42, "world", 3.14);

// 剩余参数可以是元组类型
function joinWithSeparator(separator: string, ...strings: [string, string, ...string[]]): string {
    return strings.join(separator);
}

console.log(joinWithSeparator(",", "a", "b", "c", "d")); // "a,b,c,d"
```

## 4. 函数重载

函数重载允许定义多个函数类型，根据传入参数的不同执行不同的逻辑。

### 4.1 基础函数重载

```typescript
// 重载签名
function process(value: string): string;
function process(value: number): number;
function process(value: boolean): boolean;

// 实现签名
function process(value: string | number | boolean): string | number | boolean {
    if (typeof value === "string") {
        return value.toUpperCase();
    } else if (typeof value === "number") {
        return value * 2;
    } else {
        return !value;
    }
}

console.log(process("hello"));  // "HELLO"
console.log(process(10));       // 20
console.log(process(true));     // false
```

### 4.2 不同参数数量的重载

```typescript
// 重载签名
function makeDate(timestamp: number): Date;
function makeDate(year: number, month: number, day: number): Date;

// 实现签名
function makeDate(yearOrTimestamp: number, month?: number, day?: number): Date {
    if (month !== undefined && day !== undefined) {
        return new Date(yearOrTimestamp, month, day);
    }
    return new Date(yearOrTimestamp);
}

console.log(makeDate(1234567890));           // 使用时间戳
console.log(makeDate(2024, 1, 15));          // 使用年月日
```

### 4.3 不同参数类型的重载

```typescript
// 重载签名
function getLength(value: string): number;
function getLength(value: any[]): number;
function getLength(value: { length: number }): number;

// 实现签名
function getLength(value: any): number {
    return value.length;
}

console.log(getLength("hello"));      // 5
console.log(getLength([1, 2, 3]));    // 3
console.log(getLength({ length: 10 })); // 10
```

### 4.4 返回类型不同的重载

```typescript
// 重载签名
function getUser(id: number): { id: number; name: string };
function getUser(name: string): { id: number; name: string }[];

// 实现签名
function getUser(idOrName: number | string): { id: number; name: string } | { id: number; name: string }[] {
    if (typeof idOrName === "number") {
        return { id: idOrName, name: "鱼皮" };
    }
    return [
        { id: 1, name: "鱼皮" },
        { id: 2, name: idOrName }
    ];
}

const user = getUser(1);        // { id: number; name: string }
const users = getUser("张三");   // { id: number; name: string }[]
```

### 4.5 使用接口定义重载

```typescript
// 使用接口定义函数重载
interface StringToNumber {
    (value: string): number;
    (value: string, radix: number): number;
}

const parseNumber: StringToNumber = (value: string, radix?: number): number => {
    if (radix !== undefined) {
        return parseInt(value, radix);
    }
    return parseInt(value, 10);
};

console.log(parseNumber("10"));     // 10
console.log(parseNumber("10", 2));  // 2
```

### 4.6 泛型函数重载

```typescript
// 泛型重载
function reverse<T>(array: T[]): T[];
function reverse<T>(string: string): string;

// 实现签名
function reverse<T>(value: T[] | string): T[] | string {
    if (typeof value === "string") {
        return value.split("").reverse().join("");
    }
    return [...value].reverse();
}

console.log(reverse([1, 2, 3]));      // [3, 2, 1]
console.log(reverse("hello"));        // "olleh"
```

### 4.7 重载的最佳实践

```typescript
// 1. 重载签名应该简洁明了
function formatDate(timestamp: number): string;
function formatDate(year: number, month: number, day: number): string;

function formatDate(yearOrTimestamp: number, month?: number, day?: number): string {
    let date: Date;
    if (month !== undefined && day !== undefined) {
        date = new Date(yearOrTimestamp, month, day);
    } else {
        date = new Date(yearOrTimestamp);
    }
    return date.toISOString().split("T")[0];
}

// 2. 使用联合类型简化重载（当参数类型有限时）
// 不推荐：过多的重载
function processValue(value: string): string;
function processValue(value: number): number;
function processValue(value: boolean): boolean;
function processValue(value: null): null;

// 推荐：使用联合类型
function processValue(value: string | number | boolean | null): string | number | boolean | null {
    if (typeof value === "string") {
        return value.toUpperCase();
    }
    if (typeof value === "number") {
        return value * 2;
    }
    if (typeof value === "boolean") {
        return !value;
    }
    return null;
}
```
